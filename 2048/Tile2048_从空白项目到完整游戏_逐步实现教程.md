```typescript
      .borderRadius(10)
```

绘制器不是页面状态组件，它专门封装 Canvas 绘图细节：

```typescript
export class TileRenderer {
  /** 绘制一个方块及其数字。row/col 可以是小数，用于移动动画。 */
  drawTile(context: CanvasRenderingContext2D,
    row: number, col: number, value: number,
    padding: number, gap: number, cell: number,
    scale: number, opacity: number, colorScheme: number): void {
    const size: number = cell * scale;
    const centerX: number = padding + col * (cell + gap) + cell / 2;
    const centerY: number = padding + row * (cell + gap) + cell / 2;
    const x: number = centerX - size / 2;
    const y: number = centerY - size / 2;

    context.globalAlpha = opacity;
    context.fillStyle = this.tileColor(value, colorScheme);
    this.fillRoundRect(context, x, y, size, size, size * 0.08);

    context.fillStyle = this.tileTextColor(value, colorScheme);
    context.font = this.tileFont(value, size);
    context.textAlign = 'center';
    context.textBaseline = 'middle';
    context.fillText(value.toString(), centerX, centerY);
    context.globalAlpha = 1; // 必须恢复，否则后续方块也会透明
  }

  /** Canvas 没有直接兼容所有目标版本的圆角矩形方法，因此手工构造路径。 */
  fillRoundRect(context: CanvasRenderingContext2D,
    x: number, y: number, width: number, height: number,
    radius: number): void {
    context.beginPath();
    context.moveTo(x + radius, y);
    context.arcTo(x + width, y, x + width, y + height, radius);
    context.arcTo(x + width, y + height, x, y + height, radius);
    context.arcTo(x, y + height, x, y, radius);
    context.arcTo(x, y, x + width, y, radius);
    context.closePath();
    context.fill();
  }

  /** 数字位数越多，字体越小。 */
  private tileFont(value: number, cell: number): string {
    let ratio: number = 0.45;
    if (value >= 100) {
      ratio = 0.36;
    }
    if (value >= 1000) {
      ratio = 0.28;
    }
    return 'bold ' + (cell * ratio).toFixed(1) + 'px sans-serif';
  }

  /** 根据方案编号和方块值选择颜色。 */
  private tileColor(value: number, colorScheme: number): string {
    // 海洋配色
    if (colorScheme === 1) {
      if (value === 2) return '#D9F0FA';
      if (value === 4) return '#BFE3F2';
      if (value === 8) return '#72C6E8';
      if (value === 16) return '#42ACD6';
      if (value === 32) return '#258EC2';
      if (value === 64) return '#1976A8';
      if (value === 128) return '#5B8DEF';
      if (value === 256) return '#536DCE';
      if (value === 512) return '#6657C8';
      if (value === 1024) return '#7B4BB7';
      if (value === 2048) return '#933FA4';
      return '#273C75';
    }

    // 薄荷配色
    if (colorScheme === 2) {
      if (value === 2) return '#E0F4EA';
      if (value === 4) return '#C6E9D8';
      if (value === 8) return '#8FD3B6';
      if (value === 16) return '#62BF9C';
      if (value === 32) return '#3DAA82';
      if (value === 64) return '#24906B';
      if (value === 128) return '#A8C85A';
      if (value === 256) return '#D0C84E';
      if (value === 512) return '#E5B84A';
      if (value === 1024) return '#E89A48';
      if (value === 2048) return '#E77A55';
      return '#275D50';
    }

    // 经典 2048 配色
    if (value === 2) return '#EEE4DA';
    if (value === 4) return '#EDE0C8';
    if (value === 8) return '#F2B179';
    if (value === 16) return '#F59563';
    if (value === 32) return '#F67C5F';
    if (value === 64) return '#F65E3B';
    if (value === 128) return '#EDCF72';
    if (value === 256) return '#EDCC61';
    if (value === 512) return '#EDC850';
    if (value === 1024) return '#EDC53F';
    if (value === 2048) return '#EDC22E';
    return '#3C3A32';
  }

  private tileTextColor(value: number, colorScheme: number): string {
    if (value === 2 || value === 4) {
      return colorScheme === 0 ? '#776E65' : '#35505D';
    }
    return '#F9F6F2';
  }
}
```

数据流如下：

```text
触摸 GameBoard
      ↓
GamePage 判断滑动方向
      ↓
GameEngine 执行移动、合并、计分和随机生成
      ↓
GameEngine 返回动画轨迹
      ↓
GamePage 按帧刷新 Canvas
      ↓
ScoreBoard / GameOverDialog 响应状态变化
```

```typescript
export enum GameStatus {
  PLAYING,   // 正常进行中
  WIN,       // 第一次合成 2048，显示胜利弹窗
  GAME_OVER  // 棋盘已满且不存在可合并方块

/** 用户滑动方向。 */
export enum SwipeDirection {
  UP,
  DOWN,
  LEFT,
  RIGHT
}

/** 棋盘中的一个坐标。row 为行，col 为列。 */
export interface CellPosition {
  row: number;
  col: number;
}

/**
 * 一个源方块的移动轨迹。
 * Canvas 根据 from 和 to 之间的插值绘制移动动画。
 */
export interface TileMotion {
  fromRow: number;
  fromCol: number;
  toRow: number;
  toCol: number;
  value: number;
}

/** 一次有效滑动产生的完整动画信息。 */
export interface MoveAnimation {
  motions: TileMotion[];              // 所有原方块的移动轨迹
  mergedCells: CellPosition[];        // 需要播放弹跳效果的位置
  spawnedCell: CellPosition | null;   // 新方块位置，无新方块时为 null
}
```

注意：动画数据不能只记录最终棋盘，否则页面不知道方块从哪里移动过来。

### 6.2 初始化棋盘

继续写在 `GameEngine` 类中：

```typescript
  /** 清空棋盘、分数和撤销状态，并生成两个初始方块。 */
  init(): void {
    this.grid = [];

    // 不能固定写成 [0, 0, 0, 0]，否则 5×5 和 6×6 无法工作。
    for (let row = 0; row < this.size; row++) {
      const cells: number[] = [];
      for (let col = 0; col < this.size; col++) {
        cells.push(0);
      }
      this.grid.push(cells);
    }

    this.score = 0;
    this.previousGrid = [];
    this.previousScore = 0;
    this.undoAvailable = false;
    this.clearLastMoveAnimation();

    // 经典 2048 开局有两个随机方块。
    this.spawnTile();
    this.spawnTile();
  }

  /** 切换棋盘尺寸，切换后直接开始新游戏。 */
  setSize(size: number): void {
    if (size !== 4 && size !== 5 && size !== 6) {
      return;
    }
    this.size = size;
    this.init();
  }

  getScore(): number {
    return this.score;
  }

  /** 返回副本，避免页面直接修改引擎内部棋盘。 */
  getGrid(): number[][] {
    return this.copyGrid(this.grid);
  }

  private copyGrid(source: number[][]): number[][] {
    const copy: number[][] = [];
    for (let row = 0; row < source.length; row++) {
      copy.push(source[row].slice());
    }
    return copy;
  }
```

### 6.3 实现单步撤销

```typescript
  canUndo(): boolean {
    return this.undoAvailable;
  }

  /** 恢复最近一次有效移动之前的棋盘和分数。 */
  undo(): boolean {
    if (!this.undoAvailable) {
      return false;
    }

    this.grid = this.copyGrid(this.previousGrid);
    this.score = this.previousScore;

    // 当前设计只允许撤销一步，因此恢复后立即清除快照。
    this.previousGrid = [];
    this.previousScore = 0;
    this.undoAvailable = false;
    this.clearLastMoveAnimation();
    return true;
  }
```

### 6.4 实现移动与合并

这是引擎最重要的方法：

```typescript
  /**
   * 执行一个方向的移动。
   * 返回 true 表示棋盘发生变化；只有发生变化才生成新方块。
   */
  move(direction: SwipeDirection): boolean {
    // 在修改棋盘前保存快照，供撤销使用。
    const gridBeforeMove: number[][] = this.copyGrid(this.grid);
    const scoreBeforeMove: number = this.score;

    let changed = false;
    const motions: TileMotion[] = [];
    const mergedCells: CellPosition[] = [];

    // 左右移动时 index 表示行；上下移动时 index 表示列。
    for (let index = 0; index < this.size; index++) {
      const original = this.readLine(index, direction);
      const tiles: LineTile[] = [];

      // 去掉 0，但保留每个方块的原始坐标。
      for (let offset = 0; offset < this.size; offset++) {
        if (original[offset] !== 0) {
          tiles.push({
            value: original[offset],
            position: this.getCellPosition(index, offset, direction)
          });
        }
      }

      // 创建与棋盘边长一致的全 0 结果行。
      const merged: number[] = [];
      for (let offset = 0; offset < this.size; offset++) {
        merged.push(0);
      }

      let sourceIndex = 0;
      let targetOffset = 0;

      // 按靠近滑动方向的一端开始合并。
      while (sourceIndex < tiles.length) {
        const target = this.getCellPosition(index, targetOffset, direction);
        const first = tiles[sourceIndex];

        if (sourceIndex + 1 < tiles.length &&
          first.value === tiles[sourceIndex + 1].value) {
          const second = tiles[sourceIndex + 1];
          const mergedValue = first.value * 2;

          merged[targetOffset] = mergedValue;
          this.score += mergedValue; // 分数增加合并后的数字

          // 两个源方块都移动到同一个目标位置。
          motions.push(this.createMotion(first, target));
          motions.push(this.createMotion(second, target));
          mergedCells.push({ row: target.row, col: target.col });

          // 两个方块已经参与合并，跳过两个，确保一次移动只合并一次。
          sourceIndex += 2;
        } else {
          merged[targetOffset] = first.value;
          motions.push(this.createMotion(first, target));
          sourceIndex++;
        }
        targetOffset++;
      }

      // 只有结果与原行不同，才写回棋盘。
      if (!this.sameLine(original, merged)) {
        changed = true;
        this.writeLine(index, direction, merged);
      }
    }

    if (changed) {
      // 有效移动覆盖上一次撤销快照。
      this.previousGrid = gridBeforeMove;
      this.previousScore = scoreBeforeMove;
      this.undoAvailable = true;

      // 生成一个新方块，并保存动画信息。
      const spawnedCell = this.spawnTileWithPosition();
      this.lastMoveAnimation = {
        motions: motions,
        mergedCells: mergedCells,
        spawnedCell: spawnedCell
      };
    } else {
      // 无效移动不生成方块、不覆盖撤销快照，也不播放动画。
      this.clearLastMoveAnimation();
    }

    return changed;
  }
```

`[2, 2, 2, 0]` 向左时的处理过程是：

```text
去零后：[2, 2, 2]
前两个合并：[4]
第三个不能再次参与前面的合并：[4, 2]
补零后：[4, 2, 0, 0]
```

### 6.5 实现方向坐标转换

```typescript
  private createMotion(tile: LineTile, target: CellPosition): TileMotion {
    return {
      fromRow: tile.position.row,
      fromCol: tile.position.col,
      toRow: target.row,
      toCol: target.col,
      value: tile.value
    };
  }

  /** 将“第几条线、第几个偏移”转换为真正的二维坐标。 */
  private getCellPosition(index: number, offset: number,
    direction: SwipeDirection): CellPosition {
    if (direction === SwipeDirection.LEFT) {
      return { row: index, col: offset };
    }
    if (direction === SwipeDirection.RIGHT) {
      return { row: index, col: this.size - 1 - offset };
    }
    if (direction === SwipeDirection.UP) {
      return { row: offset, col: index };
    }
    return { row: this.size - 1 - offset, col: index };
  }

  private readLine(index: number, direction: SwipeDirection): number[] {
    const line: number[] = [];
    for (let offset = 0; offset < this.size; offset++) {
      if (direction === SwipeDirection.LEFT) {
        line.push(this.grid[index][offset]);
      } else if (direction === SwipeDirection.RIGHT) {
        line.push(this.grid[index][this.size - 1 - offset]);
      } else if (direction === SwipeDirection.UP) {
        line.push(this.grid[offset][index]);
      } else {
        line.push(this.grid[this.size - 1 - offset][index]);
      }
    }
    return line;
  }

  private writeLine(index: number, direction: SwipeDirection, line: number[]): void {
    for (let offset = 0; offset < this.size; offset++) {
      if (direction === SwipeDirection.LEFT) {
        this.grid[index][offset] = line[offset];
      } else if (direction === SwipeDirection.RIGHT) {
        this.grid[index][this.size - 1 - offset] = line[offset];
      } else if (direction === SwipeDirection.UP) {
        this.grid[offset][index] = line[offset];
      } else {
        this.grid[this.size - 1 - offset][index] = line[offset];
      }
    }
  }

  private sameLine(first: number[], second: number[]): boolean {
    for (let index = 0; index < this.size; index++) {
      if (first[index] !== second[index]) {
        return false;
      }
    }
    return true;
  }
```

通过反向读取右移和下移的数据，合并算法只需要写一次。

### 6.6 随机生成方块

```typescript
  spawnTile(): boolean {
    return this.spawnTileWithPosition() !== null;
  }

  private spawnTileWithPosition(): CellPosition | null {
    const emptyCells: number[][] = [];

    for (let row = 0; row < this.size; row++) {
      for (let col = 0; col < this.size; col++) {
        if (this.grid[row][col] === 0) {
          emptyCells.push([row, col]);
        }
      }
    }

    if (emptyCells.length === 0) {
      return null;
    }

    // 随机选择一个空位。
    const location = emptyCells[Math.floor(Math.random() * emptyCells.length)];

    // 90% 概率生成 2，10% 概率生成 4。
    this.grid[location[0]][location[1]] = Math.random() < 0.9 ? 2 : 4;
    return { row: location[0], col: location[1] };
  }
```

### 6.7 胜利和游戏结束判断

```typescript
  hasWon(): boolean {
    for (let row = 0; row < this.size; row++) {
      for (let col = 0; col < this.size; col++) {
        if (this.grid[row][col] >= 2048) {
          return true;
        }
      }
    }
    return false;
  }

  isGameOver(): boolean {
    for (let row = 0; row < this.size; row++) {
      for (let col = 0; col < this.size; col++) {
        const value = this.grid[row][col];

        // 还有空格，一定没有结束。
        if (value === 0) {
          return false;
        }

        // 与下方相同，仍可以纵向合并。
        if (row + 1 < this.size && value === this.grid[row + 1][col]) {
          return false;
        }

        // 与右侧相同，仍可以横向合并。
        if (col + 1 < this.size && value === this.grid[row][col + 1]) {
          return false;
        }
      }
    }
    return true;
  }
```

只检查右侧和下方即可覆盖所有相邻关系，不必再重复检查左侧和上方。

```typescript
import dataPreferences from '@ohos.data.preferences';

/** 只负责最高分的读取和保存。 */
class ScoreManager {
  private preferences: dataPreferences.Preferences | null = null;
  private readonly preferenceName: string = 'tile2048_preferences';
  private readonly bestScoreKey: string = 'best_score';

  /** 应用启动时传入 UIAbilityContext，打开 Preferences 文件。 */
  async configure(context: Context): Promise<void> {
    this.preferences = await dataPreferences.getPreferences(
      context,
      this.preferenceName
    );
  }

  async getBestScore(): Promise<number> {
    if (this.preferences === null) {
      return 0;
    }
    return await this.preferences.get(this.bestScoreKey, 0) as number;
  }

  async saveBestScore(score: number): Promise<void> {
    if (this.preferences === null) {
      return;
    }

    const current = await this.getBestScore();
    if (score > current) {
      await this.preferences.put(this.bestScoreKey, score);

      // flush 确保数据写入磁盘，应用重启后仍然存在。
      await this.preferences.flush();
    }
  }
}

/** 导出单例，避免页面重复创建 Preferences 管理器。 */
export const scoreManager: ScoreManager = new ScoreManager();
```

```typescript
  private processSwipe(direction: SwipeDirection): void {
    // 弹窗显示或动画进行时，不接受新的滑动。
    if (this.gameStatus !== GameStatus.PLAYING || this.isAnimating) {
      return;

    // 无效方向直接返回，不生成新方块、不加分。
```

### 9.5 实现触摸方向识别

```typescript
  private handleOffset(deltaX: number, deltaY: number): void {
    if (this.panHandled) {
      return;
    }

    // 两个方向都未达到阈值，认为是轻点或短滑动。
    if (Math.abs(deltaX) < SWIPE_THRESHOLD &&
      Math.abs(deltaY) < SWIPE_THRESHOLD) {
      return;
    }

    this.panHandled = true;

    // 选择绝对位移更大的轴作为滑动方向。
    if (Math.abs(deltaX) > Math.abs(deltaY)) {
      this.processSwipe(deltaX > 0
        ? SwipeDirection.RIGHT
        : SwipeDirection.LEFT);
    } else {
      this.processSwipe(deltaY > 0
        ? SwipeDirection.DOWN
        : SwipeDirection.UP);
    }
  }

  private handleBoardTouch(event: TouchEvent): void {
    if (event.type === TouchType.Down && event.touches.length > 0) {
      this.panHandled = false;
      this.touchStartX = event.touches[0].x;
      this.touchStartY = event.touches[0].y;
      return;
    }

    if (event.type === TouchType.Move && event.touches.length > 0) {
      this.handleOffset(
        event.touches[0].x - this.touchStartX,
        event.touches[0].y - this.touchStartY
      );
      return;
    }

    if (event.type === TouchType.Up && event.changedTouches.length > 0) {
      this.handleOffset(
        event.changedTouches[0].x - this.touchStartX,
        event.changedTouches[0].y - this.touchStartY
      );
    }
  }
```

### 9.6 实现移动、合并和出现动画

```typescript
  private playMoveAnimation(animation: MoveAnimation): void {
    if (animation.motions.length === 0) {
      this.finishMove();
      return;
    }

    this.stopAnimation();
    this.isAnimating = true;

    const startedAt: number = Date.now();

    // speed 越大，实际持续时间越短。
    const moveDuration: number = MOVE_ANIMATION_MS / this.animationSpeed;
    const effectDuration: number = EFFECT_ANIMATION_MS / this.animationSpeed;

    this.animationTimer = setInterval(() => {
      const elapsed: number = Date.now() - startedAt;

      if (elapsed < moveDuration) {
        const progress = this.easeOutCubic(elapsed / moveDuration);
        this.drawMovingTiles(animation, progress);
        return;
      }

      const effectProgress = Math.min(
        1,
        (elapsed - moveDuration) / effectDuration
      );
      this.drawTileEffects(animation, effectProgress);

      if (effectProgress >= 1) {
        this.stopAnimation();
        this.finishMove();
      }
    }, FRAME_INTERVAL_MS);
  }

  private stopAnimation(): void {
    if (this.animationTimer >= 0) {
      clearInterval(this.animationTimer);
      this.animationTimer = -1;
    }
    this.isAnimating = false;
  }

  /** 三次方缓出：开始较快，接近终点时减速。 */
  private easeOutCubic(progress: number): number {
    return 1 - Math.pow(1 - progress, 3);
  }
```

### 9.7 绘制最终棋盘

```typescript
  private drawBoard(): void {
    if (!this.boardReady) {
      return;
    }

    const width: number = this.ctx.width;
    const height: number = this.ctx.height;
    if (width <= 0 || height <= 0) {
      return;
    }

    const padding: number = width * 0.025;
    const gap: number = width * 0.025;
    const cell: number = (
      width - padding * 2 - gap * (this.boardSize - 1)
    ) / this.boardSize;

    this.drawGridBackground(width, height, padding, gap, cell);

    const grid: number[][] = this.engine.getGrid();
    for (let row = 0; row < this.boardSize; row++) {
      for (let col = 0; col < this.boardSize; col++) {
        const value: number = grid[row][col];
        if (value !== 0) {
          this.drawTile(row, col, value, padding, gap, cell, 1, 1);
        }
      }
    }
  }

  private drawGridBackground(width: number, height: number,
    padding: number, gap: number, cell: number): void {
    this.ctx.globalAlpha = 1;
    this.ctx.clearRect(0, 0, width, height);
    this.ctx.fillStyle = this.emptyCellColor();

    for (let row = 0; row < this.boardSize; row++) {
      for (let col = 0; col < this.boardSize; col++) {
        const x = padding + col * (cell + gap);
        const y = padding + row * (cell + gap);
        this.tileRenderer.fillRoundRect(
          this.ctx, x, y, cell, cell, cell * 0.08
        );
      }
    }
  }

  private drawTile(row: number, col: number, value: number,
    padding: number, gap: number, cell: number,
    scale: number, opacity: number): void {
    this.tileRenderer.drawTile(
      this.ctx,
      row,
      col,
      value,
      padding,
      gap,
      cell,
      scale,
      opacity,
      this.colorScheme
    );
  }
```

`drawMovingTiles()` 的原则是清空棋盘后绘制所有源方块的插值位置；`drawTileEffects()` 的原则是先绘制普通最终方块，再单独绘制合并位置和新方块。完整实现请直接对照当前 `GamePage.ets` 中这两个方法。

合并与新方块阶段绘制：

```typescript
  private drawTileEffects(animation: MoveAnimation, progress: number): void {
    if (!this.boardReady) {
      return;
    }

    const width: number = this.ctx.width;
    const height: number = this.ctx.height;
    const padding: number = width * 0.025;
    const gap: number = width * 0.025;
    const cell: number = (
      width - padding * 2 - gap * (this.boardSize - 1)
    ) / this.boardSize;
    const grid: number[][] = this.engine.getGrid();

    this.drawGridBackground(width, height, padding, gap, cell);

    // 先绘制普通最终方块，合并位置和新方块留到后面单独绘制。
    for (let row = 0; row < this.boardSize; row++) {
      for (let col = 0; col < this.boardSize; col++) {
        const value = grid[row][col];
        if (value !== 0 &&
          !this.isCellIn(row, col, animation.mergedCells) &&
          !this.isSameCell(row, col, animation.spawnedCell)) {
          this.drawTile(row, col, value, padding, gap, cell, 1, 1);
        }
      }
    }

    // 1 → 1.16 → 1，形成合并弹跳。
    const bounceScale: number =
      1 + Math.sin(progress * Math.PI) * 0.16;
    for (let merged of animation.mergedCells) {
      this.drawTile(
        merged.row,
        merged.col,
        grid[merged.row][merged.col],
        padding,
        gap,
        cell,
        bounceScale,
        1
      );
    }

    // 新方块从 0.2 倍逐渐放大，同时透明度从 0 变到 1。
    if (animation.spawnedCell !== null) {
      const spawned = animation.spawnedCell;
      const appearScale: number =
        0.2 + 0.8 * this.easeOutCubic(progress);
      this.drawTile(
        spawned.row,
        spawned.col,
        grid[spawned.row][spawned.col],
        padding,
        gap,
        cell,
        appearScale,
        progress
      );
    }
  }

  private isCellIn(row: number, col: number,
    cells: CellPosition[]): boolean {
    for (let cell of cells) {
      if (cell.row === row && cell.col === col) {
        return true;
      }
    }
    return false;
  }

  private isSameCell(row: number, col: number,
    cell: CellPosition | null): boolean {
    return cell !== null && cell.row === row && cell.col === col;
  }
```

## 12. 分阶段验收

不要等全部代码写完才测试。推荐每完成一层就运行一次。

### 阶段一：引擎

- 初始棋盘只有两个 2 或 4。
- `[2,2,4,4]` 左移得到 `[4,8,0,0]`，分数增加 12。
- `[2,2,2,0]` 左移得到 `[4,2,0,0]`。
- 无效滑动不生成新方块。
- 4×4、5×5、6×6 都不会数组越界。

### 阶段二：手势和 Canvas

- 四个方向都能识别。
- 轻点和小于 30px 的滑动不触发。
- 数字居中，网格间距均匀。
- 6×6 时格子自动缩小但不重叠。

### 阶段三：动画

- 方块平滑移动到目标位置。
- 两个方块合并后有放大回弹。
- 新方块从小到大并逐渐显现。
- 动画期间连续滑动不会破坏棋盘。
- 慢速、标准、快速三档有明显差异。

### 阶段四：状态和持久化

- 合并后 SCORE 正确增加。
- 超过 BEST 后立即更新。

- 经典、海洋、薄荷配色即时生效。
- 切换棋盘尺寸后开始一局新游戏。

## 13. 常见错误

### 13.1 Canvas 中内容不更新

Canvas 内部绘制不是普通声明式组件。修改引擎后必须主动调用：

```typescript
this.drawBoard();
```

### 13.2 无效移动也生成方块

必须检查 `engine.move()` 的返回值：

```typescript
if (!this.engine.move(direction)) {
  return;
}
```

### 13.3 一次移动中同一个方块重复合并

合并成功后必须执行：

```typescript
sourceIndex += 2;
```

不能只增加 1。

### 13.4 自定义组件属性无法传入

需要从父组件传入的回调不能声明为 `private`。另外不要使用 `backgroundColor` 作为自定义字段名，它可能与 ArkUI 的系统属性冲突，可改成：

```typescript
@Prop boardColor: string;
@Prop panelColor: string;
```

### 13.5 页面路径错误

下面三处必须对应：

```text
实际文件：view/page/GamePage.ets
main_pages.json：view/page/GamePage
loadContent：view/page/GamePage
```
